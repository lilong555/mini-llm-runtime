cmake_minimum_required(VERSION 3.24)
project(LLMServe VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/MSVCIncludes.cmake)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
endif()

option(LLMSERVE_WITH_LLAMA "Build the llama.cpp model backend" ON)
option(LLMSERVE_CUDA "Enable llama.cpp CUDA execution" OFF)
option(LLMSERVE_BUILD_TESTS "Build LLMServe tests" ON)
set(LLMSERVE_LLAMA_COMMIT "911f6cdc8ab8a530b2bee09ee61471a6f3178eeb")

find_package(Threads REQUIRED)
add_library(minillm_compute
    src/minillm/kernels.cpp
    src/minillm/paged_kv.cpp
    src/minillm/parallel.cpp
)
target_include_directories(minillm_compute PUBLIC include)
target_link_libraries(minillm_compute PUBLIC Threads::Threads)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64|X86_64|i.86")
    target_sources(minillm_compute PRIVATE src/minillm/kernels_avx2.cpp)
    target_compile_definitions(minillm_compute PUBLIC MINILLM_HAS_AVX2=1)
    if(MSVC)
        set_source_files_properties(src/minillm/kernels_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    else()
        set_source_files_properties(src/minillm/kernels_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c")
    endif()
endif()
if(MSVC)
    target_compile_options(minillm_compute PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(minillm_compute PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_library(llmserve_core
    src/config.cpp
    src/block_pool.cpp
    src/prefix_index.cpp
    src/scheduler.cpp
    src/utf8.cpp
    src/engine.cpp
)
target_include_directories(llmserve_core PUBLIC include)
target_link_libraries(llmserve_core PUBLIC Threads::Threads)
target_compile_features(llmserve_core PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(llmserve_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(llmserve_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

if(LLMSERVE_WITH_LLAMA)
    set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp")
    if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
        message(FATAL_ERROR "Fetch the pinned llama.cpp checkout. See README.md.")
    endif()
    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
        WORKING_DIRECTORY "${LLAMA_SOURCE_DIR}"
        OUTPUT_VARIABLE LLAMA_ACTUAL_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
        COMMAND_ERROR_IS_FATAL ANY)
    if(NOT LLAMA_ACTUAL_COMMIT STREQUAL LLMSERVE_LLAMA_COMMIT)
        message(FATAL_ERROR "llama.cpp must be pinned to ${LLMSERVE_LLAMA_COMMIT}")
    endif()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
    set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
    set(GGML_CUDA ${LLMSERVE_CUDA} CACHE BOOL "" FORCE)
    set(CMAKE_CXX_STANDARD 17)
    add_subdirectory(third_party/llama.cpp EXCLUDE_FROM_ALL)
    set(CMAKE_CXX_STANDARD 20)
    add_library(llmserve_llama src/llama_runner.cpp)
    target_link_libraries(llmserve_llama PUBLIC llmserve_core PRIVATE llama)
    if(MSVC)
        target_compile_options(llmserve_llama PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(llmserve_llama PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_library(minillm_runtime src/minillm/gguf_model.cpp src/minillm/runtime.cpp)
    target_link_libraries(minillm_runtime PUBLIC minillm_compute PRIVATE llama)
    add_library(llmserve_mini src/mini_runner.cpp)
    target_link_libraries(llmserve_mini PUBLIC llmserve_core minillm_runtime)
    if(MSVC)
        target_compile_options(minillm_runtime PRIVATE /W4 /permissive- /utf-8)
        target_compile_options(llmserve_mini PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(minillm_runtime PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
        target_compile_options(llmserve_mini PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
    add_executable(mini-llm apps/mini_main.cpp)
    target_link_libraries(mini-llm PRIVATE minillm_runtime llmserve_core llama vendor::nlohmann)
    add_library(llmserve_http_vendor STATIC "${LLAMA_SOURCE_DIR}/vendor/cpp-httplib/httplib.cpp")
    target_include_directories(llmserve_http_vendor SYSTEM PUBLIC "${LLAMA_SOURCE_DIR}/vendor/cpp-httplib")
    set_target_properties(llmserve_http_vendor PROPERTIES CXX_STANDARD 17)
    target_link_libraries(llmserve_http_vendor PRIVATE Threads::Threads)
    if(WIN32)
        target_link_libraries(llmserve_http_vendor PUBLIC ws2_32 crypt32)
    endif()
    add_library(llmserve_http src/http_server.cpp)
    target_link_libraries(llmserve_http PUBLIC llmserve_core PRIVATE llmserve_http_vendor vendor::nlohmann)
    add_executable(llmserve apps/server_main.cpp)
    target_link_libraries(llmserve PRIVATE llmserve_http llmserve_mini llmserve_llama llama)
    add_executable(llmserve-bench apps/bench_main.cpp)
    target_link_libraries(llmserve-bench PRIVATE llmserve_http_vendor vendor::nlohmann)
    add_executable(mini-kernel-bench apps/kernel_bench.cpp)
    target_link_libraries(mini-kernel-bench PRIVATE minillm_compute vendor::nlohmann)
    add_executable(mini-kv-cache-bench apps/kv_cache_bench.cpp)
    target_link_libraries(mini-kv-cache-bench PRIVATE minillm_compute vendor::nlohmann)
endif()

if(LLMSERVE_BUILD_TESTS)
    enable_testing()
    add_executable(llmserve-unit-tests tests/core_tests.cpp)
    target_link_libraries(llmserve-unit-tests PRIVATE llmserve_core minillm_compute)
    add_test(NAME unit COMMAND llmserve-unit-tests)
    set_tests_properties(unit PROPERTIES TIMEOUT 30)
    if(LLMSERVE_WITH_LLAMA)
        add_executable(llmserve-model-tests tests/model_tests.cpp)
        target_link_libraries(llmserve-model-tests PRIVATE llmserve_mini llmserve_llama llama vendor::nlohmann)
        add_executable(llmserve-http-tests tests/http_tests.cpp)
        target_link_libraries(llmserve-http-tests PRIVATE llmserve_http_vendor vendor::nlohmann)
        add_executable(llmserve-gguf-tests tests/gguf_tests.cpp)
        target_link_libraries(llmserve-gguf-tests PRIVATE minillm_runtime llama)
        add_test(NAME gguf COMMAND llmserve-gguf-tests)
        set_tests_properties(gguf PROPERTIES TIMEOUT 30)
    endif()
endif()
