if(LLMSERVE_WITH_LLAMA)
    set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp")
    if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
        message(FATAL_ERROR "Fetch the pinned llama.cpp checkout. See README.md.")
    endif()
    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
        WORKING_DIRECTORY "${LLAMA_SOURCE_DIR}"
        OUTPUT_VARIABLE LLAMA_ACTUAL_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
        COMMAND_ERROR_IS_FATAL ANY)
    if(NOT LLAMA_ACTUAL_COMMIT STREQUAL LLMSERVE_LLAMA_COMMIT)
        message(FATAL_ERROR "llama.cpp must be pinned to ${LLMSERVE_LLAMA_COMMIT}")
    endif()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
    set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
    set(GGML_CUDA ${LLMSERVE_CUDA} CACHE BOOL "" FORCE)
    set(CMAKE_CXX_STANDARD 17)
    add_subdirectory(third_party/llama.cpp EXCLUDE_FROM_ALL)
    set(CMAKE_CXX_STANDARD 20)
    add_library(llmserve_llama src/llama_runner.cpp)
    target_link_libraries(llmserve_llama PUBLIC llmserve_core PRIVATE llama)
    if(MSVC)
        target_compile_options(llmserve_llama PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(llmserve_llama PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_library(minillm_model
        src/minillm/gguf_model.cpp
        src/minillm/qwen3_model.cpp
        src/minillm/tokenizer.cpp
    )
    target_link_libraries(minillm_model PUBLIC minillm_compute PRIVATE llama)
    if(MINILLM_ENABLE_CUDA)
        target_link_libraries(minillm_cuda PRIVATE minillm_model vendor::hash)
    endif()
    add_library(minillm_runtime src/minillm/runtime.cpp)
    target_link_libraries(minillm_runtime PUBLIC minillm_model minillm_compute)
    add_library(llmserve_mini src/mini_runner.cpp)
    target_link_libraries(llmserve_mini PUBLIC llmserve_core minillm_runtime)
    if(MINILLM_ENABLE_CUDA)
        add_library(llmserve_mini_cuda src/mini_cuda_runner.cpp)
        target_link_libraries(llmserve_mini_cuda PUBLIC llmserve_core PRIVATE minillm_cuda)
        if(MSVC)
            target_compile_options(llmserve_mini_cuda PRIVATE /W4 /permissive- /utf-8)
        else()
            target_compile_options(llmserve_mini_cuda PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
        endif()
    endif()
    if(MSVC)
        target_compile_options(minillm_model PRIVATE /W4 /permissive- /utf-8)
        target_compile_options(minillm_runtime PRIVATE /W4 /permissive- /utf-8)
        target_compile_options(llmserve_mini PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(minillm_model PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
        target_compile_options(minillm_runtime PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
        target_compile_options(llmserve_mini PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
    add_executable(mini-llm apps/mini_main.cpp)
    target_link_libraries(mini-llm PRIVATE minillm_runtime llmserve_core llama vendor::nlohmann)
    if(MINILLM_ENABLE_CUDA)
        add_executable(mini-cuda-llm apps/cuda_main.cpp)
        target_link_libraries(mini-cuda-llm PRIVATE minillm_cuda llmserve_core llama vendor::nlohmann vendor::hash)

        target_include_directories(mini-cuda-kernel-bench PRIVATE src/minillm/cuda)
        target_link_libraries(mini-cuda-kernel-bench PRIVATE minillm_cuda minillm_model vendor::nlohmann vendor::hash)
    endif()
    add_library(llmserve_http_vendor STATIC "${LLAMA_SOURCE_DIR}/vendor/cpp-httplib/httplib.cpp")
    target_include_directories(llmserve_http_vendor SYSTEM PUBLIC "${LLAMA_SOURCE_DIR}/vendor/cpp-httplib")

    target_link_libraries(llmserve_http_vendor PRIVATE Threads::Threads)

    endif()

    target_link_libraries(llmserve PRIVATE llmserve_http llmserve_mini llmserve_llama llama vendor::nlohmann)

    add_executable(mini-runtime-bench apps/runtime_bench.cpp)
